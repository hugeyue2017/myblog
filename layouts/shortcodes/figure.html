{{- /* ─────────────────────────────────────────────────────────────────────────── 带图注的图片：figure + figcaption 覆盖主题原版。 为什么必须覆盖：主题原版是直接 ， 而 Markdown 里的 ![]() 走的是 render-image 钩子 —— 只有钩子会把图片 重新编码、抹掉 EXIF、生成 WebP 与四档 srcset。 用主题原版写 figure，等于绕过整条流水线： 访客下的是原图，而且原图的 GPS / 拍摄时间会一起发出去。 所以这里复用同一套逻辑：多档 srcset + WebP + 绝不外发原图。 ─────────────────────────────────────────────────────────────────────────── */ -}} {{- $src := .Get "src" -}} {{- $alt := .Get "alt" | default (.Get "caption") | default "" -}} {{- $caption := .Get "caption" -}} {{- $u := urls.Parse $src -}} {{- $img := "" -}} {{- if not $u.IsAbs -}} {{- $clean := strings.TrimPrefix "/" (strings.TrimPrefix "./" $u.Path) -}} {{- $img = or (.Page.Resources.Get $clean) (resources.Get $clean) -}} {{- end -}} {{- $class := .Get "class" -}} {{- if eq (.Get "align") "center" -}}{{- $class = printf "align-center %s" $class -}}{{- end -}} {{- if $img -}} {{- $widths := slice 480 800 1200 1800 -}} {{- $maxW := $img.Width -}} {{- $jpgSet := slice -}} {{- $webpSet := slice -}} {{- $srcURL := "" -}} {{- range $widths -}} {{- if le . $maxW -}} {{- $w := . -}} {{- $r := $img.Resize (printf "%dx q86 Lanczos" $w) -}} {{- $jpgSet = $jpgSet | append (printf "%s %dw" $r.RelPermalink $w) -}} {{- $srcURL = $r.RelPermalink -}} {{- if hugo.IsExtended -}} {{- $webpSet = $webpSet | append (printf "%s %dw" ($img.Resize (printf "%dx webp q80 Lanczos" $w)).RelPermalink $w) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* 走不到 480 档的小图（截图 / 裁剪图）单独重新编码一份， 确保发出去的文件一定不带元数据 */ -}} {{- if not $jpgSet -}} {{- $r := $img.Process "jpeg q86" -}} {{- $jpgSet = $jpgSet | append (printf "%s %dw" $r.RelPermalink $maxW) -}} {{- $srcURL = $r.RelPermalink -}} {{- if hugo.IsExtended -}} {{- $webpSet = $webpSet | append (printf "%s %dw" ($img.Process "webp q80").RelPermalink $maxW) -}} {{- end -}} {{- end -}} {{- if $webpSet }} {{- end }} {{- else -}} {{- warnf "figure: 未在 assets/ 找到 %q（页面 %s）—— 原图请放到 assets/images/ 下" $src .Page.File.Path -}} {{- end -}} {{- with $caption }} {{- end }}
